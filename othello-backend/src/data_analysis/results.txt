AvgSelectedHyperparamPerWhite = 
VAR SelectedHyperparam = SELECTEDVALUE(HyperparamSelector[HyperparamName]) 
VAR WinningBlackModels = 
    FILTER(
        FactGame, 
        FactGame[Winner] = "black" && FactGame[WhiteModelID] = SELECTEDVALUE(FactGame[WhiteModelID])
    )

VAR WinningBlackModelIDs = 
    SELECTCOLUMNS(WinningBlackModels, "BlackModelID", FactGame[BlackModelID])

RETURN
    AVERAGEX(
        FILTER(
            DimModel, 
            DimModel[ModelID] IN WinningBlackModelIDs
        ),
        SWITCH(
            SelectedHyperparam,
            "Mobility", DimModel[MobilityWeight],
            "Stability", DimModel[StabilityWeight],
            "Coin", DimModel[CoinWeight],
            "Corners", DimModel[CornersWeight],
            "DangerZones", DimModel[DangerZonesWeight],
            "Edges", DimModel[EdgesWeight],
            "Wedges", DimModel[WedgesWeight],
            BLANK()
        )
    )

--------------

BlackModelWinrates = 

VAR GamesPlayed =
    CALCULATE(
        COUNTROWS(FactGame),
        FactGame[WhiteModelID] = SELECTEDVALUE(FactGame[WhiteModelID])
    )

VAR BlackWins =
    CALCULATE(
        COUNTROWS(FactGame),
        FactGame[WhiteModelID] = SELECTEDVALUE(FactGame[WhiteModelID]),
        FactGame[Winner] = "black"
    )

RETURN
    DIVIDE(BlackWins, GamesPlayed, 0)

--------------

WhiteModelWinrates = 

VAR GamesPlayed =
    CALCULATE(
        COUNTROWS(FactGame),
        FactGame[WhiteModelID] = SELECTEDVALUE(FactGame[WhiteModelID])
    )

VAR WhiteWins =
    CALCULATE(
        COUNTROWS(FactGame),
        FactGame[WhiteModelID] = SELECTEDVALUE(FactGame[WhiteModelID]),
        FactGame[Winner] = "white"
    )

RETURN
    DIVIDE(WhiteWins, GamesPlayed, 0)
